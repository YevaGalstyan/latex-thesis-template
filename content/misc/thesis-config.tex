% =============================================================
%  THESIS CONFIGURATION
%  This is the only file you need to edit to personalize the
%  template. Everything else picks these values up.
% =============================================================

% --- Document language ---------------------------------------
% Main language of the thesis, passed to babel.
% Examples: english, british, ngerman, french, spanish
% (MacTeX includes all languages; on Linux you may need the
%  matching texlive-lang-* package)
\newcommand{\thesislanguage}{english}

% --- Title ---------------------------------------------------
\newcommand{\thesistitle}{Title of Your Thesis}
\newcommand{\thesissubtitle}{An Optional Subtitle}   % leave {} empty to hide

% --- Author --------------------------------------------------
\newcommand{\thesisauthor}{Your Name}
\newcommand{\thesisstudentid}{1234567}               % leave {} empty to hide

% --- Degree --------------------------------------------------
% e.g. Bachelor's Thesis / Master's Thesis / Doctoral Dissertation
\newcommand{\thesistype}{Master's Thesis}
% e.g. Bachelor of Science / Master of Science / Doctor of Philosophy
\newcommand{\thesisdegree}{Master of Science}
\newcommand{\thesisprogram}{Your Study Program}

% --- Institution ---------------------------------------------
\newcommand{\thesisuniversity}{Your University}
\newcommand{\thesisdepartment}{Department of Your Field}
% Path to your university logo (PNG or PDF), relative to main.tex.
% Leave {} empty to hide the logo.
\newcommand{\thesislogo}{}                           % e.g. assets/logo.png

% --- Supervision ---------------------------------------------
\newcommand{\thesissupervisor}{Prof. Dr. First Supervisor}
\newcommand{\thesiscosupervisor}{Dr. Second Supervisor}  % leave {} empty to hide

% --- Submission ----------------------------------------------
\newcommand{\thesisdate}{\today}                     % or e.g. {25 September 2026}
\newcommand{\thesislocation}{City}                   % used in the declaration
